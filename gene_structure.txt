<allele> ::=  0|1
<subject> ::= 7*7<allele>  ; allows for 128 subjects, there are currently 80 A-levels, 20 T-levels and 14 BTECs
<classroom> ::= 7*7<allele>  ; allows for 128 classrooms, at national classroom size limits this means that up to 3840 students can be in lesson at once
<teacher> ::= 8*8<allele>  ; allows for 256 teachers, this ensures that there will always be enough teachers per classroom
<class> ::= 4*4<allele>  ; gives 16 classes available per subject
<timeslot> ::= 6*6<allele>  ; 64 timeslots available. maximum of 12 lessons per day. Limits can be applied to exclude some values during next generation generation
<gene> ::= <class><subject><timeslot><teacher><classroom> ; maximum of 2^32 combinations FITS WITHIN UINT32 :)
